Q7. File and Directory Tests
==============================================================================
Script: file_check.sh
----------------------------------------
#!/bin/bash
# file_check.sh - regular file / directory / neither
if [ -z "$1" ]; then
    echo "Usage: $0 <path>"
    exit 1
fi
if [ -f "$1" ]; then
    echo "$1 is a regular file."
    if [ -x "$1" ]; then
        echo "It is executable."
    else
        echo "It is NOT executable."
    fi
elif [ -d "$1" ]; then
    echo "$1 is a directory."
else
    echo "$1 is neither a regular file nor a directory (or does not exist)."
fi
----------------------------------------

Q13. Output Redirection
==============================================================================
Commands:
  $ ls data > output/files.txt        # > overwrites the file
  $ cat output/files.txt
  $ ls data >> output/files.txt       # >> appends to the file
  $ cat output/files.txt
  $ echo "Overwritten!" > output/files.txt   # > again - old content lost
  $ cat output/files.txt

$ ls data > output/files.txt
$ cat output/files.txt
names.txt
notes.txt
students.txt
$ ls data >> output/files.txt
$ cat output/files.txt
names.txt
notes.txt
students.txt
names.txt
notes.txt
students.txt
$ echo 'Overwritten!' > output/files.txt
$ cat output/files.txt
Overwritten!

Difference: '>' creates/overwrites (old content lost); '>>' appends to the end.

Q19. Backup Checker Script
==============================================================================
Script: backup_check.sh
----------------------------------------
#!/bin/bash
# backup_check.sh <directory>
if [ -z "$1" ] || [ ! -d "$1" ]; then
    echo "Error: '$1' is not a valid directory."
    echo "Usage: $0 <directory>"
    exit 1
fi
count=0
for f in "$1"/*.bak
do
    if [ -f "$f" ]; then
        echo "Backup found: $f"
        count=$((count + 1))
    fi
done
echo "Total .bak files: $count"
if [ "$count" -eq 0 ]; then
    echo "$(date '+%F %T') - No backup files found in $1" >> logs/backup_log.txt
fi
----------------------------------------
Make executable:  $ chmod +x backup_check.sh

$ ./backup_check.sh backups
Backup found: backups/db_01.bak
Backup found: backups/db_02.bak
Total .bak files: 2

$ ./backup_check.sh data
Total .bak files: 0

$ cat logs/backup_log.txt
2026-11-03 02:39:38 - No backup files found in data

$ ./backup_check.sh nodir; echo "Exit: $?"

Usage: ./backup_check.sh <directory>

Q20. Integrated File Report
==============================================================================
Script: file_report.sh
----------------------------------------
#!/bin/bash
# file_report.sh <filename>
if [ $# -ne 1 ]; then
    echo "Usage: $0 <filename>"
    exit 1
fi
file="$1"
if [ ! -e "$file" ]; then
    echo "Error: '$file' does not exist."
    echo "$(date '+%F %T') - FAILED: $file not found" >> logs/execution_history.log
    exit 2
fi
if [ ! -r "$file" ]; then
    echo "Error: '$file' is not readable."
    echo "$(date '+%F %T') - FAILED: $file not readable" >> logs/execution_history.log
    exit 3
fi
echo "Report for: $file"
echo "Lines      : $(wc -l < "$file")"
echo "Words      : $(wc -w < "$file")"
echo "Characters : $(wc -m < "$file")"
echo "$(date '+%F %T') - Processed: $file" >> logs/execution_history.log
----------------------------------------
Make executable:  $ chmod +x file_report.sh

$ ./file_report.sh data/notes.txt
Report for: data/notes.txt
Lines      : 3
Words      : 9
Characters : 64

$ ./file_report.sh missing.txt; echo "Exit: $?"
Error: 'missing.txt' does not exist.
Exit: 2

$ ./file_report.sh; echo "Exit: $?"
Usage: ./file_report.sh <filename>
